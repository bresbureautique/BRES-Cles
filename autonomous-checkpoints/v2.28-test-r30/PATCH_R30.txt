function recoverOCRCandidates(raw){
  const s=(raw||"").toUpperCase();
  const chunks=[];
  const re=/[A-Z0-9]+/g;
  let m;
  while((m=re.exec(s))) chunks.push({text:m[0],start:m.index,end:re.lastIndex});
  const weakGap=/^[ \t.\-‐‑‒–—−﹣－\u00A0\u202F]{1,3}$/u;
  const out=[];
  for(let i=0;i<chunks.length;i++){
    let compact="",totalGap=0;
    for(let j=i;j<Math.min(chunks.length,i+8);j++){
      if(j>i){
        const gap=s.slice(chunks[j-1].end,chunks[j].start);
        if(!weakGap.test(gap)) break;
        totalGap+=gap.length;
        if(totalGap>3) break;
      }
      compact+=chunks[j].text;
      if(compact.length>10) break;
      if(/^[A-Z]{1,4}\d{1,3}[A-Z]$/.test(compact)) out.push(compact);
    }
  }
  return [...new Set(out)];
}
